\section{Có gì mới}

\subsection*{Phiên bản v0.7}
\begin{itemize}
  \item Bổ sung hộp thoại chỉnh sửa trực quan cho mọi loại khối: bảng, danh
    sách, công thức, mục, hình ảnh, định lý, ghi chú và mã nguồn.
  \item Ra mắt bảng ký hiệu toán học mở nhanh bằng phím tắt
    \phim{Ctrl}+\phim{K} \phim{M}.
  \item Thêm nút ⋯ trong bảng Mục lục để chèn nhanh tham chiếu, nhân bản, xóa
    khối hoặc sắp xếp lại vị trí các mục.
  \item Hỗ trợ tìm kiếm mọi khối trên toàn dự án bằng \phim{Ctrl}+\phim{P} kết
    hợp tiền tố \texttt{!}.
  \item Cho phép kéo thả để đổi vị trí cột trong bảng, cũng như dùng phím tắt
    \phim{Alt}+\phim{←} / \phim{→} trực tiếp trong mã nguồn.
\end{itemize}
